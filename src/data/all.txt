// ============================================
// 1. AUTHENTICATION & ROLES
// ============================================

model Role {
  id            String   @id @default(uuid())
  name          String   @unique
  descriptionEn String?  @map("description_en")
  descriptionAm String?  @map("description_am")
  users         User[]
  createdAt     DateTime @default(now()) @map("created_at")

  @@map("roles")
}

model User {
  id             String    @id @default(uuid())
  fullName       String    @map("full_name")
  email          String    @unique
  passwordHash   String    @map("password_hash")
  sessionVersion Int       @default(1)
  roleId         String    @map("role_id")
  role           Role      @relation(fields: [roleId], references: [id])

  isActive      Boolean   @default(true) @map("is_active")
  deactivatedAt DateTime? @map("deactivated_at")

  lastLoginAt DateTime? @map("last_login_at")
  createdAt   DateTime  @default(now()) @map("created_at")
  updatedAt   DateTime  @updatedAt @map("updated_at")

  refreshTokens       RefreshToken[]
  passwordResetTokens PasswordResetToken[]

  galleryImages        GalleryImage[] @relation("galleryUploader")
  galleryImagesApproved GalleryImage[] @relation("galleryApprover")
  galleryImagesDeleted GalleryImage[] @relation("galleryDeleter")

  contactMessagesAssigned ContactMessage[] @relation("assignedMessages")
  contactMessagesReplied  ContactMessage[] @relation("repliedMessages")
  contactMessagesDeleted  ContactMessage[] @relation("contactMessageDeleter")

  newsArticlesAuthor  NewsArticle[] @relation("newsAuthor")
  newsArticlesCreated NewsArticle[] @relation("newsCreator")
  newsArticlesUpdated NewsArticle[] @relation("newsUpdater")
  newsArticlesDeleted NewsArticle[] @relation("newsDeleter")

  servicesCreated Service[] @relation("serviceCreator")
  servicesUpdated Service[] @relation("serviceUpdater")
  servicesDeleted Service[] @relation("serviceDeleter")

  partnersCreated Partner[] @relation("partnerCreator")
  partnersUpdated Partner[] @relation("partnerUpdater")
  partnersDeleted Partner[] @relation("partnerDeleter")

  branchesCreated Branch[] @relation("branchCreator")
  branchesUpdated Branch[] @relation("branchUpdater")
  branchesDeleted Branch[] @relation("branchDeleter")

  reportsUploaded Report[] @relation("reportUploader")
  reportsDeleted  Report[] @relation("reportDeleter")

  notifications           Notification[]
  siteSettingsUpdated     SiteSettings[]     @relation("siteSettingsUpdater")

  organizationStatsCreated OrganizationStat[] @relation("statCreator")
  organizationStatsUpdated OrganizationStat[] @relation("statUpdater")

  historyMilestonesCreated HistoryMilestone[] @relation("milestoneCreator")
  historyMilestonesUpdated HistoryMilestone[] @relation("milestoneUpdater")
  historyMilestonesDeleted HistoryMilestone[] @relation("milestoneDeleter")

  testimonialsCreated Testimonial[] @relation("testimonialCreator")
  testimonialsUpdated Testimonial[] @relation("testimonialUpdater")
  testimonialsDeleted Testimonial[] @relation("testimonialDeleter")

  boardMembersCreated BoardMember[] @relation("boardMemberCreator")
  boardMembersUpdated BoardMember[] @relation("boardMemberUpdater")
  boardMembersDeleted BoardMember[] @relation("boardMemberDeleter")

  managementMembersCreated ManagementTeamMember[] @relation("managementMemberCreator")
  managementMembersUpdated ManagementTeamMember[] @relation("managementMemberUpdater")
  managementMembersDeleted ManagementTeamMember[] @relation("managementMemberDeleter")

  impactMetricsCreated ImpactMetric[] @relation("metricCreator")
  impactMetricsUpdated ImpactMetric[] @relation("metricUpdater")
  impactMetricsDeleted ImpactMetric[] @relation("metricDeleter")

  auditLogs AuditLog[]

  @@index([roleId])
  @@index([isActive])
  @@map("users")
}

// ============================================
// 2. AUTH SUPPORT: refresh tokens & password resets
// ============================================

model RefreshToken {
  id        String    @id @default(uuid())
  userId    String    @map("user_id")
  user      User      @relation(fields: [userId], references: [id], onDelete: Cascade)
  tokenHash String    @unique @map("token_hash")
  userAgent String?   @map("user_agent")
  ipAddress String?   @map("ip_address")
  expiresAt DateTime  @map("expires_at")
  revokedAt DateTime? @map("revoked_at")
  createdAt DateTime  @default(now()) @map("created_at")

  @@index([userId])
  @@index([expiresAt])
  @@map("refresh_tokens")
}

model PasswordResetToken {
  id        String    @id @default(uuid())
  userId    String    @map("user_id")
  user      User      @relation(fields: [userId], references: [id], onDelete: Cascade)
  tokenHash String    @unique @map("token_hash")
  expiresAt DateTime  @map("expires_at")
  usedAt    DateTime? @map("used_at")
  createdAt DateTime  @default(now()) @map("created_at")

  @@index([userId])
  @@map("password_reset_tokens")
}

  ipAddress String?  @map("ip_address")
  createdAt DateTime @default(now()) @map("created_at")

  @@index([userId])
  @@index([tableName, recordId])
  @@index([createdAt])
  @@map("audit_logs")
}

// 17. SERVICES & SERVICE CATEGORIES
// ============================================

model ServiceCategory {
  id            String    @id @default(uuid())
  nameEn        String    @unique @map("name_en")
  nameAm        String    @unique @map("name_am")
  descriptionEn String?   @map("description_en")
  descriptionAm String?   @map("description_am")
  services      Service[]
  createdAt     DateTime  @default(now()) @map("created_at")

  @@map("service_categories")
}

model Service {
  id            String          @id @default(uuid())
  slug          String          @unique
  titleEn       String          @map("title_en")
  titleAm       String          @map("title_am")
  descriptionEn String          @map("description_en")
  descriptionAm String          @map("description_am")
  icon          String?
  imagePath     String?         @map("image_path")
  categoryId    String?         @map("category_id")
  category      ServiceCategory? @relation(fields: [categoryId], references: [id], onDelete: SetNull)
  displayOrder  Int             @default(0) @map("display_order")
  isActive      Boolean         @default(true) @map("is_active")

  createdBy String?   @map("created_by")
  creator   User?     @relation("serviceCreator", fields: [createdBy], references: [id], onDelete: SetNull)
  updatedBy String?   @map("updated_by")
  updater   User?     @relation("serviceUpdater", fields: [updatedBy], references: [id], onDelete: SetNull)
  createdAt DateTime  @default(now()) @map("created_at")
  updatedAt DateTime  @updatedAt @map("updated_at")
  deletedAt DateTime? @map("deleted_at")
  deletedBy String?   @map("deleted_by")
  deleter   User?     @relation("serviceDeleter", fields: [deletedBy], references: [id], onDelete: SetNull)

  @@index([categoryId])
  @@index([displayOrder])
  @@index([isActive])
  @@map("services")
}
